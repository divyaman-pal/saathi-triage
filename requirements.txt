# Modelling

# scipy is a transitive dependency of scikit-learn and is pinned EXPLICITLY:
# left to resolve on its own it picks a build that caps numpy below 2.3, which
# then conflicts with the numpy pin below. 1.16.2 supports numpy 2.3 and has
# cp314 wheels. An unpinned transitive dependency is what broke this deploy.
scipy==1.16.2
# numpy 2.3.2 is the earliest release with cp314 wheels. Streamlit Cloud now
# defaults to Python 3.14, and on 2.2.6 pip falls back to the source tarball and
# builds numpy from scratch on one CPU. Every pin in this file must have a
# prebuilt wheel for 3.11 through 3.14 - check before lowering any of them.
numpy==2.3.2

# NOTE: the `shap` package is deliberately NOT a dependency. Feature attribution
# uses xgboost's own exact TreeSHAP via `pred_contribs=True` (see
# core/risk_model.py), so pulling in the shap library would add a slow, heavy
# install for code that never imports it.

# Operator surfaces
streamlit==1.62.0

# Bounded LLM integration. The system runs fully without this installed or
# configured - the deterministic template renderer takes over and every
# safety-path output is unchanged.
